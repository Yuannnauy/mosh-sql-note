# 第七章 内置函数

## 数值函数
-- 四舍五入
```sql
select round(5.73)
```
-- 指定精度 比如保留一位小数
```sql
select round(5.73，1)
```
-- 截断数字
```sql
select truncate(5.7385,2)  会得到5.73
```
-- 上限函数 返回大于等于该数字的整数：
```sql
select ceiling(5.2)  #得到6
```
-- floor地板函数：返回小于等于该数字的整数：
```sql
select floor(5.7)  #得到5
```
-- 绝对值
```sql
select abs(-5.7)
```
-- 生成0-1区间的随机浮点数
```sql
select rand()
```

## 字符串函数
```SQL
-- LENGTH()  返回字符串的长度
SELECT LENGTH('SKY');       -- 3
-- UPPER()   全部变为大写字母
SELECT UPPER('hello');      -- HELLO
-- LOWER()   全部变为小写字母
SELECT LOWER('WORLD');      -- world
-- LTRIM()  移除字符串左边的空白字符或其他预定义字符
SELECT LTRIM('  Sky  ');    -- 'Sky  ' 
-- RTRIM()  移除字符串右边的空白字符或其他预定义字符
SELECT RTRIM('  Sky  ');    -- '  Sky' 
-- TRIM()  移除字符串两边的空白字符或其他预定义字符
SELECT RTRIM('  Sky  ');    -- 'Sky' 
-- LEFT(string, n)  获取字符串左边n个字符
SELECT LEFT('Kindergaten', 4);    -- Kind
-- RIGHT(string, n)  获取字符串右边n个字符
SELECT RIGHT('Kindergaten', 4);    -- aten
-- SUBSTRING(string, start, len)  获取字符串(包含)start位置开始的len个字符
SELECT SUBSTRING('Kindergaten', 4, 3);  -- der
-- len可省略，省略后会从start位置开始算到最后一个字符：
SELECT SUBSTRING('Kindergaten', 4);  -- dergaten
-- LOCATE(substring, string)  返回查找到substring的第一个字符（串）在string中的位置
SELECT LOCATE('der', 'Kindergaten');    -- 4
-- 查询不到会返回0
SELECT LOCATE('que', 'Kindergaten');    -- 0
-- REPLACE(string, substring,replacestring)  将字符串中，substring替换为replacestring
SELECT REPLACE('Kindergaten', 'gaten','garden'); -- Kindergarden
-- CONCAT(string1, string2)  连接两个字符串
SELECT CONCAT('Kinder', 'garden');    -- Kindergarden
-- eg. 连接 姓和名，组成完整的姓名
USE sql_store;
SELECT CONCAT(first_name,' ',last_name) AS full_name
FROM customers;
```

##  mysql中的日期函数
```sql
SELECT 
	NOW(),      # 当前的日期和时间
	CURDATE(),  # 当前的日期，没有时间
	CURTIME();  # 当前的时间，没有日期
```

```sql
-- 从日期中提取特定的值    
SELECT YEAR(NOW()),  # 提取 年份
SELECT MONTH(NOW()), #      月份
SELECT DAY(NOW()),   #      日期
SELECT HOUR(NOW()),  #      时
SELECT MINUTE(NOW()),#      分
SELECT SECOND(NOW());#      秒
```

```sql
-- 返回字符串名称
select DAYNAME(NOW()),   # 星期几 eg:monday, wednesday
select MONTHNAME(NOW()); # 月份全称 eg:march, June
```

```sql
-- EXTRACT()("提取") 函数用于返回日期/时间的单独部分，
-- 比如年、月、日、小时、分钟等等，可用于mysql以外的数据库
SELECT EXTRACT(YEAR FROM NOW())
SELECT EXTRACT(MONTH FROM NOW())
SELECT EXTRACT(DAY FROM NOW())
SELECT EXTRACT(YEAR_MONTH FROM NOW())
-- exercise：查询所有当前年份的订单：
use sql_store;
select * 
from orders
where year(order_date)=year(now())
```

## 格式化日期和时间
| 格式 | 描述 |
| :--- | :--- |
| `%a` | 缩写星期名 |
| `%b` | 缩写月名 |
| `%c` | 月，数值 |
| `%D` | 带有英文前缀的月中的天 |
| `%d` | 月的天，数值(00-31) |
| `%e` | 月的天，数值(0-31) |
| `%f` | 微秒 |
| `%H` | 小时 (00-23) |
| `%h` | 小时 (01-12) |
| `%I` | 小时 (01-12) |
| `%i` | 分钟，数值(00-59) |
| `%j` | 年的天 (001-366) |
| `%k` | 小时 (0-23) |
| `%l` | 小时 (1-12) |
| `%M` | 月名 |
| `%m` | 月，数值(00-12) |
| `%p` | AM 或 PM |
| `%r` | 时间，12-小时（hh:mm:ss AM 或 PM） |
| `%S` | 秒(00-59) |
| `%s` | 秒(00-59) |
| `%T` | 时间, 24-小时 (hh:mm:ss) |
| `%U` | 周 (00-53) 星期日是一周的第一天 |
| `%u` | 周 (00-53) 星期一是一周的第一天 |
| `%V` | 周 (01-53) 星期日是一周的第一天，与 %X 使用 |
| `%v` | 周 (01-53) 星期一是一周的第一天，与 %x 使用 |
| `%W` | 星期名 |
| `%w` | 周的天 （0=星期日, 6=星期六） |
| `%X` | 年，其中的星期日是周的第一天，4 位，与 %V 使用 |
| `%x` | 年，其中的星期一是周的第一天，4 位，与 %v 使用 |
| `%Y` | 年，4 位 |
| `%y` | 年，2 位 |
```sql
-- 举例：
select date_format(now(),%M %d %Y)
-- 得到：June 3 2026
select time_format(now(),'%H: %i %p')
-- 得到：12：58 AM
```

## 计算日期和时间
```sql
-- 计算日期和时间
-- DATE_ADD DATE_SUB
-- 在当前日期加上1天
SELECT DATE_ADD(NOW(), INTERVAL 1 DAY);

-- 在当前日期减去1年
SELECT DATE_ADD(NOW(), INTERVAL -1 YEAR);
SELECT DATE_SUB(NOW(), INTERVAL 1 YEAR);

-- 计算两个日期天数的间隔  DATEDIFF：前一个减后一个
SELECT DATEDIFF('2019-01-01','2019-01-05');#得到-4
SELECT DATEDIFF('2019-01-05','2019-01-01');#得到4

-- 计算秒数间隔：返回从零点计算的秒数
SELECT TIME_TO_SEC('9:00') #得到32400
SELECT TIME_TO_SEC('9:00')- TIME_TO_SEC('9:02'); #得到-120
```

## ifnull和coalesce函数
把shipper_id中的null值替换为not assigned
```sql
select
order_id,
ifnull(shipper_id,'not assigned') as shipper
from orders
--用coalesce函数写 coalesce：返回列举值中的第一个非空值
select
order_id,
ifnull(shipper_id, comments, 'not assigned') as shipper
from orders
```

## case运算符
if只需判断一个条件 case用来判断多个条件
```sql
select 
order_id
case
when year(order_date)=year(now()) then 'active'
when year(order_date)=year(now())-1 then 'last year'
when year(order_date)<year(now())-1 then 'archive'
else 'future'
end as category
-- exercise
select 
concat(first_name,' ',last_name) as customer,
points,
case 
when points>3000 then 'gold'
when points >=2000 then 'silver'
else 'bronze'
end as category
from customers
order by points desc
```
