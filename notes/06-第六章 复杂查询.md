# 复杂查询

## 子查询

where内可以写select子查询
avg()必须在select中用 因此不能：
```sql
where salary>avg(salary)
-- 应该：
where salary>
(select avg(salary) from employees)
```

## in运算符
```sql
-- 选择没有出现在订单中的product
use sql_store;
select name
from products 
where product_id not in(
select distinct product_id
from order_items)
```

## 子查询 vs 连接

```sql
-- 用in运算符查询没有发票的客户：
use sql_invoicing;
select *
from clients
where client_id not in (
select distinct client_id
from invoices)
-- 也可用左连接查询：
use sql_invoicing;
select *
from clients
left join invoices using(client_id)
where incoice_id is not null
-- 选择执行时间更短，可读性强的查法
```

## ALL 关键字
```sql
-- client 3的发票不止一张 想要查询大于client的全部发票：
select *
from invoices
where invoices_total>(
select max(invoice_total)
from invoices
where client_id=3
-- 用all关键字：
select *
from invoices
where invoice_total>all (
select invoice_total
from invoices
where client_id=3
)
-- all（1,2,3,...）如果invoice_total大于all()括号内的所有值 则会被返回
-- select max() 可以和 all（）互换
```

## any， some关键字
```sql
-- 找到至少有两张发票的客户：
select *
from clients
where client_id in(
select client_id
from invoices
group by client_id
having count(*)>2
)

-- 另一种写法：
select *
from clients
where client_id = any(
select client_id
from invoices
group by client_id
having count(*)>2
)
```

## 相关子查询

```sql
-- 查询每个办公室薪资高于该办公室平均薪资的员工：
use sql_hr;
select * 
from emplyees e
where salary>
(
select avg(salary)
from employees
where office_id=e.office_id
)
-- 会比较慢
```
